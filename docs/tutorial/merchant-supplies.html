function potion_stacks(stat) { return character.items.map((item, slot) => ({ item, slot })) .filter(({ item }) => item && !item.l && !item.b && !item.r && G.items[item.name].type === "pot" && G.items[item.name].gives.some(give => give[0] === stat && give[1] > 0)); } function potion_count(stat) { return potion_stacks(stat).reduce((sum, entry) => sum + (entry.item.q || 1), 0); } async function supply_friend() { if (character.ctype !== "merchant" || character.rip) return; var own = get_characters().map(c => c.name); var party = get_party() || {}; var name = Object.keys(party).find(name => name !== character.name && own.includes(name) && !party[name].rip && party[name].type !== "merchant" && G.maps[party[name].map] && !G.maps[party[name].map].instance); if (!name) return show_json({ delivered: false, reason: "no_party_recipient" }); var hp = Math.max(0, 40 - potion_count("hp")); var mp = Math.max(0, 40 - potion_count("mp")); var cost = hp * G.items.hpot0.g + mp * G.items.mpot0.g; if (character.gold < cost + 4000) return; if (hp || mp) { await smart_move(find_npc("fancypots")); if (hp) await buy_with_gold("hpot0", hp); if (mp) await buy_with_gold("mpot0", mp); } var destination = (get_party() || {})[name]; if (!destination || destination.rip || G.maps[destination.map].instance) return; await smart_move({ map: destination.map, x: destination.x, y: destination.y }); for (var stat of ["hp", "mp"]) { var remaining = 20; while (remaining > 0) { var friend = get_player(name), entry = potion_stacks(stat)[0]; if (!friend || friend.rip || friend.map !== character.map || distance(character, friend) > 200 || !entry) return; var amount = Math.min(remaining, entry.item.q || 1, potion_count(stat) - 20); if (amount <= 0) return; await send_item(name, entry.slot, amount); remaining -= amount; } } show_json({ delivered: true, name }); } supply_friend().catch(show_json);

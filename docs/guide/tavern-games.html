"check" : to_call <= 2 * data.blinds[1] ? "call" : "fold"); } catch (error) { show_json(error); } finally { busy = false; } }); try { await poker_join(table.buyin[0], seat); joined = true; } catch (error) { game.remove(listener); throw error; } } // Sitting out keeps the seat and the stack while you step away; sit back in to be dealt again: // await poker_sit_out(); ... await poker_sit_in(); // A restart or a closed Tavern rejects requests with reason "tavern_closing"; standing too far from a stool with "poker_stool_far". // This spends gold: // play_poker(5).catch(show_json);

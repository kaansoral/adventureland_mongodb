async function graveyard_rising() { var rising = server.status.graveyard; if (!rising || !rising.live) return; var target = get_nearest_monster({ type: "gargoyle" }) || get_nearest_monster({ type: "ghoul" }) || get_nearest_monster({ type: "zombie" }); if (character.map != "spookytown" || !target) return await join("graveyard"); if (!is_in_range(target)) return await move(character.x + (target.x - character.x) / 2, character.y + (target.y - character.y) / 2); if (can_attack(target)) return await attack(target); } graveyard_rising().catch(show_json);

async function ride_with_the_horseman() { var ride = server.status.horseman; if (!ride || !ride.live) return; if (character.map != "halloween") return await join("horseman"); var horseman = get_nearest_monster({ type: "horseman" }); if (!horseman) return await smart_move({ map: "halloween", x: ride.x, y: ride.y }); if (!is_in_range(horseman)) return await move(character.x + (horseman.x - character.x) / 2, character.y + (horseman.y - character.y) / 2); if (can_attack(horseman)) return await attack(horseman); } ride_with_the_horseman().catch(show_json);

function point_distance(a, b) { var dx = b.x - a.x; var dy = b.y - a.y; return Math.sqrt(dx * dx + dy * dy); }

function point_at_distance(a, b, units) { var angle = Math.atan2(b.y - a.y, b.x - a.x); return { x: a.x + Math.cos(angle) * units, y: a.y + Math.sin(angle) * units }; }
